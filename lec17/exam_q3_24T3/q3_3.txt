mod plugins;

// These types implement the `Visitor` trait
use plugins::{QuerySelectorTransformer, ServerSideTransformer};

struct Transformer {
    plugins: Vec<Plugin>,
}

struct Plugin {
	// change this to dyn vvv
    visitor: Box<dyn Visitor>,
}

trait Visitor {
    fn visit(&self, source: String) -> String;
}

pub fn transform_code(code: String) -> String {
    let transformer = Transformer {
        plugins: vec![
            Plugin {
				// wrap in a Box vvv
                visitor: Box::new(QuerySelectorTransformer),
            },
            Plugin {
				// wrap in a Box vvv
                visitor: Box::new(ServerSideTransformer),
            },
        ],
    };

    let transformed_code = transformer.run(code);

    transformed_code
}

impl Transformer {
    fn run(self, mut code: String) -> String {
        for plugin in self.plugins {
            code = plugin.visitor.visit(code);
        }

        code
    }
}
